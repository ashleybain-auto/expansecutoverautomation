CurCount = 0
Do
CurCount = CurCount + 1
CurChar = Left(Right(CurDirectory, CurCount), 1)
Loop Until CurCount >= Len(CurDirectory) Or CurChar = Application.PathSeparator

CurDirectory = Left(CurDirectory, Len(CurDirectory) - CurCount) & Application.PathSeparator

    Set MyObject = New Scripting.FileSystemObject

    'Search for files that we want to replace code module in
            Set wbTarget = ActiveWorkbook
            
            'First we need to remove the existing module

            CurModulePath = myFile.Path

               LCase(Right(CurModuleName, 4)) = ".bas" Then

        ModPresent = True

            wbTarget.VBProject.VBComponents.Import ModulePath

Sub ModifyServerName(CurServerName As String)

    If Sheets("Settings").Range("settingServer").Value <> CurServerName Then

        Application.Calculation = xlCalculationManual
        Application.Cursor = xlWait
        Application.ScreenUpdating = False
        
        Sheets("Settings").Visible = True
        
        With Sheets("Settings").Range("settingServer").Validation
            .Delete
            .Add Type:=xlValidateInputOnly, AlertStyle:=xlValidAlertStop, Operator _
            :=xlBetween
            .IgnoreBlank = True
            .InCellDropdown = True
            .ShowInput = True
            .ShowError = True
        End With
        
        Sheets("Settings").Range("settingServer").Value = CurServerName
        
        Sheets("Settings").Visible = False
        
        Application.Calculation = xlCalculationAutomatic
        Application.Cursor = xlDefault
        Application.ScreenUpdating = True
    
    End If

End Sub

Sub PrepareAccommodationsSheet()
    
    Dim CurWkBk As Workbook
    Set CurWkBk = ActiveWorkbook
    Dim Cur As Worksheet
    Dim NewSheet As Worksheet
    Dim TestSheetName As String
    On Error Resume Next
    TestSheetName = CurWkBk.Sheets("Accommodations").Name
    On Error GoTo 0
    If TestSheetName <> "Accommodations" Then
        Application.DisplayAlerts = False
        Set NewSheet = Sheets.Add
        NewSheet.Name = "Accommodations"
        With NewSheet.Tab
            .ThemeColor = xlThemeColorAccent1
            .TintAndShade = 0.799981688894314
        End With
        Set Cur = NewSheet
        Application.DisplayAlerts = True
    Else
        Set Cur = CurWkBk.Sheets(TestSheetName)
        Dim myShape As Shape
        For Each myShape In Cur.Shapes
            If myShape.Type = msoControlButton Then
                myShape.Delete
            End If
        Next myShape
    End If
    
    CurBtnRow = 1
    AddButton CurWkBk, Cur, Cur.Range("A" & CurBtnRow & ":A" & CurBtnRow + 1), "Btn_Accommodations_Home", "Home", "Link_To_Home"

Sub PushUpdates_ErrorCategories()

    Dim Cur As Worksheet
    Set Cur = Sheets("ErrorCategories")

    Dim dUpdateItem As New Scripting.Dictionary
    Dim CurErrorField As String
    Dim CurFieldDisplayNameSource As String
    Dim CurFieldDisplayNameTarget As String
    Dim CurErrorCategory As String
    Dim CurActionToTake As String
    
    MaxRows = Cur.Range("tblErrorCategories").Rows.Count
    For i = 1 To MaxRows
        CurActionToTake = Trim(Cur.Range("tblErrorCategories[Action To Take]").Rows(i))
            'Update
            If CurActionToTake = "Update" Then
                CurErrorField = Trim(Cur.Range("tblErrorCategories[Discrepancy_FieldName]").Rows(i))
                
           '     CurKey = CurErrorField
                
                CurFieldDisplayNameSource = Trim(Cur.Range("tblErrorCategories[Discrepancy_FieldName]").Rows(i))
                CurFieldDisplayNameTarget = Trim(Cur.Range("tblErrorCategories[Discrepancy_FieldName]").Rows(i))
                CurErrorCategory = Trim(Cur.Range("tblErrorCategories[Discrepancy_Category]").Rows(i))
                
         '       CurValue = CurErrorCategory
                 Dim CurValues(1 To 3) As String
                    
                    CurValues(1) = CurFieldDisplayNameSource
                    CurValues(2) = CurFieldDisplayNameTarget
                    CurValues(3) = CurErrorCategory
                
                If dUpdateItem.Exists(CurErrorField) = False Then
'                    dUpdateItem.Add CurKey, CurValues
                    dUpdateItem.Add CurErrorField, CurValues
                End If
            End If
          
    Next i
    
    If dUpdateItem.Count > 0 Then
        Call UpdateMapping_ErrorCategories(dUpdateItem)
    End If


    Call RefreshData_ErrorCategories

Private Sub ApplyDataValidation_Services(Cur As Worksheet)

    MaxRows = Cur.Range("tblServices").Rows.Count

    'Change Cell colors to Green to indicate NOT to disturb the cells
    Call Utility_ApplyFormating_UneditableFields(Cur, Cur.Range("tblServices[[FacilityMnem_Source]:[FacilityMnem_Target]]"))

                                                    , Cur.Range("tblServices[ServiceMnem_Target]") _
                                                )

     'Add Top/Left Alignment

Sub PushUpdates_Services()

    Dim dUpdateItem As New Scripting.Dictionary

    Dim CurFacTar As String
    Dim CurServicesMnemSource As String
    
    
    Dim CurServicesMnemTarget As String
    Dim CurServicesNameTarget As String
    Dim CurServicesPAcodeTarget As String
    Dim CurActiveFlagTarget As String

    Dim CurVisibleToUsersTarget As String
    Dim CurExcludeFromUsersTarget As String

    MaxRows = Cur.Range("tblServices").Rows.Count
    For i = 1 To MaxRows
        CurActionToTake = Trim(Cur.Range("tblServices[Action To Take]").Rows(i))
            'Update
            If CurActionToTake = "Update" Then
                CurFacSrc = Trim(Cur.Range("tblServices[FacilityMnem_Source]").Rows(i))
                CurFacTar = Trim(Cur.Range("tblServices[FacilityMnem_Target]").Rows(i))

                CurKey = CurFacSrc & vbTab & CurFacTar & vbTab & CurServicesMnemSource

                CurExcludeFromUsersTarget = Trim(Cur.Range("tblServices[ExcludeFromSharepoint]").Rows(i))

                           CurServicesPAcodeTarget & vbTab & CurActiveFlagTarget & vbTab & _

End Sub

Sub SortWorksheetsTabs(CurWkBk As Workbook)
Application.ScreenUpdating = False
Dim ShCount As Integer, i As Integer, j As Integer
ShCount = CurWkBk.Sheets.Count

For i = 1 To ShCount - 1
    For j = i + 1 To ShCount
        If UCase(CurWkBk.Sheets(j).Name) < UCase(CurWkBk.Sheets(i).Name) Then
            CurWkBk.Sheets(j).Move before:=CurWkBk.Sheets(i)
        End If
    Next j
Next i

CurWkBk.Sheets("Table of Contents").Move before:=CurWkBk.Sheets(1)
Application.ScreenUpdating = True
End Sub
