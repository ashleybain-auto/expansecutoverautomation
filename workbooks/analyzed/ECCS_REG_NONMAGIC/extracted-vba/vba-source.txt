Private Sub Workbook_Open()

MsgBox "Click OK to begin Refreshing Spreadsheet."
Application.Cursor = xlWait

Messagebox = 6
'Messagebox = MsgBox("Do you want to update user forms?", vbYesNo, "Update Yes/No")
If Messagebox = 6 Then

Dim lngFileCounter As Long, wbTarget As Workbook
Dim myWorkbook As FileSystemObject
Application.ScreenUpdating = False
On Error Resume Next

CurDirectory = "\\xrdcnpsvmdpt01_cfs_ag1.hca.corpad.net\SREShare\Expanse\ACOE_MTX\ExcelTool_REG_Cutover_ConfigTools\NonMagic_ConfigTool\book1.xlsb"
CurCount = 0
Do
CurCount = CurCount + 1
CurChar = Left(Right(CurDirectory, CurCount), 1)
Loop Until CurCount >= Len(CurDirectory) Or CurChar = Application.PathSeparator

If CurChar <> Application.PathSeparator Then
Exit Sub
End If

CurDirectory = Left(CurDirectory, Len(CurDirectory) - CurCount) & Application.PathSeparator
mySourcePath = CurDirectory
    Set MyObject = New Scripting.FileSystemObject
    Set mySource = MyObject.GetFolder(mySourcePath)

For Each myFile In mySource.Files

    'Search for files that we want to replace code module in
            Set wbTarget = ActiveWorkbook
            
            'First we need to remove the existing module
            CurModuleName = myFile.Name
            CurModulePath = myFile.Path
            
            If LCase(Right(CurModuleName, 4)) = ".frm" Or _
                LCase(Right(CurModuleName, 4)) = ".cls" Or _
                LCase(Right(CurModuleName, 4)) = ".bas" Then
            ModuleName = Left(CurModuleName, Len(CurModuleName) - 4)
            ModulePath = CurModulePath
            DoEvents
'            On Error Resume Next
    Do
            ModPresent = True
            Application.EnableEvents = False
            wbTarget.VBProject.VBComponents.Remove wbTarget.VBProject.VBComponents(ModuleName)
            DoEvents
            Application.EnableEvents = True
            ModPresent = False
    On Error Resume Next
    '- MODULE
    Set MyModule = ActiveWorkbook.VBProject.VBComponents(ModuleName).CodeModule
    If Err.Number = 0 Then
        ModPresent = True
    End If
    Loop Until ModPresent = False
'            On Error GoTo 0
            'Now we can import the new module, save the workbook and close.
            wbTarget.VBProject.VBComponents.Import ModulePath
            End If
            
Next myFile




    Dim VBAEditor As VBIDE.VBE
    Dim vbProj As VBIDE.VBProject
    Dim chkRef As VBIDE.Reference
    Dim BoolExists As Boolean
   ' Refer to the activedocument's VBA project.
    Set CurRefArray = CreateObject("Scripting.Dictionary")
    Set VBAEditor = Application.VBE
    Set vbProj = ActiveWorkbook.VBProject
    CurRefArray.Add "Scripting", "C:\Windows\SysWOW64\scrrun.dll"
    CurRefArray.Add "SHDocVw", "C:\Windows\SysWOW64\ieframe.dll"
    CurRefArray.Add "ADODB", "C:\Program Files (x86)\Common Files\System\ado\msado26.tlb"
    CurRefArray.Add "Access", "C:\Program Files (x86)\Microsoft Office\Office15\MSACC.OLB"
    CurRefArray.Add "DAO", "C:\Program Files (x86)\Common Files\microsoft shared\DAO\dao360.dll"
    
    For Each myKey In CurRefArray
    DoEvents
    For Each chkRef In vbProj.References
    DoEvents
        If chkRef.Name = myKey Then
            BoolExists = True
            GoTo CleanUp
        End If
    Next chkRef
    PathArray = CurRefArray(myKey)
    vbProj.References.AddFromFile PathArray
CleanUp:
    
    Next myKey

    Set vbProj = Nothing
    Set VBAEditor = Nothing

End If

Application.Run ("RunUpdatesForCurrentVersion")

Sub ModifyServerName(CurServerName As String)

    If Sheets("Settings").Range("settingServer").Value <> CurServerName Then

        Application.Calculation = xlCalculationManual
        Application.Cursor = xlWait
        Application.ScreenUpdating = False
        
        Sheets("Settings").Visible = True
        
        With Sheets("Settings").Range("settingServer").Validation
            .Delete
            .Add Type:=xlValidateInputOnly, AlertStyle:=xlValidAlertStop, Operator _
            :=xlBetween
            .IgnoreBlank = True
            .InCellDropdown = True
            .ShowInput = True
            .ShowError = True
        End With
        
        Sheets("Settings").Range("settingServer").Value = CurServerName
        
        Sheets("Settings").Visible = False
        
        Application.Calculation = xlCalculationAutomatic
        Application.Cursor = xlDefault
        Application.ScreenUpdating = True
    
    End If

End Sub

Sub PrepareStateSheet()
    
    Dim CurWkBk As Workbook
    Set CurWkBk = ActiveWorkbook
    Dim Cur As Worksheet
    Dim NewSheet As Worksheet
    Dim TestSheetName As String
    On Error Resume Next
    TestSheetName = CurWkBk.Sheets("State").Name
    On Error GoTo 0
    If TestSheetName <> "State" Then
        Application.DisplayAlerts = False
        Set NewSheet = Sheets.Add
        NewSheet.Name = "State"
        With NewSheet.Tab
            .ThemeColor = xlThemeColorAccent1
            .TintAndShade = 0.799981688894314
        End With
        Set Cur = NewSheet
        Application.DisplayAlerts = True
    Else
        Set Cur = CurWkBk.Sheets(TestSheetName)
        Dim myShape As Shape
        For Each myShape In Cur.Shapes
            If myShape.Type = msoControlButton Then
                myShape.Delete
            End If
        Next myShape
    End If
    
    CurBtnRow = 1
    AddButton CurWkBk, Cur, Cur.Range("A" & CurBtnRow & ":A" & CurBtnRow + 1), "Btn_State_Home", "Home", "Link_To_Home"
    AddButton CurWkBk, Cur, Cur.Range("E" & CurBtnRow & ":F" & CurBtnRow + 1), "Btn_State_ProcessUpdate", "Process Update(s)", "PushUpdates_State"

End Sub

Sub RefreshData_State()

     ' In VBE you need to go Tools References and check Microsoft Active X Data Objects 2.x library
     '
     Dim CurFacSrc As String
     CurFacSrc = "*ALL*"
     
    Set CurWkBk = ActiveWorkbook

    Dim Cur As Worksheet

                    Chr(13) & "  - Facility Mnem : {FAC}" & Chr(13) & _
                    Chr(9) & "e.g. (CMCH, COCBM)" & Chr(13) & _
                    Chr(13) & "  - Wave and Facility: {WAVE}*{FAC}-{FAC}" & Chr(13) & _
                    Chr(9) & "e.g. (CMC*CMCH, 2D*COCBM-COCCL)" & Chr(13) & _

    End If

    Call OpenConnection

    Dim Rs As ADODB.Recordset
    Set Rs = New ADODB.Recordset
    
    Dim Cmd As New ADODB.Command
    Set Cmd = Nothing
    Cmd.CommandText = "[dbo].[UI_Excel_Select_Xwalk_NonMagic_State]"
    Cmd.CommandType = adCmdStoredProc
    Cmd.ActiveConnection = Cn
    Set prm = Cmd.CreateParameter("@facmnem_src", adVarChar, adParamInput, 25, id_FacilityMnem)
    Cmd.Parameters.Append (prm)
    Set prm = Cmd.CreateParameter("@usersec34", adVarChar, adParamInput, 255, CurUser34)
    Cmd.Parameters.Append (prm)
    
    Set Rs = Cmd.Execute
    
    Cur.Cells.Clear
    Cur.Cells.Clear
    Call PrepareStateSheet
    Cur.Columns.NumberFormat = "@"
    
    For i = 0 To Rs.Fields.Count - 1
        Cur.Cells(3, i + 1) = Rs.Fields(i).Name
    Next i

    Cur.Range("A4").CopyFromRecordset Rs
    
    Cur.ListObjects.Add(xlSrcRange, Cur.Range(Cur.Cells(3, 1), Cur.Cells(Application.WorksheetFunction.CountA(Cur.Range("A:A")) + 2, Application.WorksheetFunction.CountA(Cur.Range("3:3")))), , xlYes).Name = "tblState"
    Cur.ListObjects("tblState").TableStyle = "TableStyleMedium15"
    Cur.Select
    Cur.Cells(1, 1).Select
    Cur.Cells(1, 3) = "Excel Refreshed: " & Now
    Cur.Cells(2, 3) = ""
    CurWkBk.RefreshAll
    Cur.Columns.AutoFit
    
    If Rs.State <> 0 Then Rs.Close
    Call CloseConnection
    
    Call ApplyDataValidation_State(Cur)
        
End Sub

Sub PushUpdates_State()

    Dim Cur As Worksheet
    Set Cur = Sheets("State")

    Dim dUpdateItem As New Scripting.Dictionary
    Dim CurFacTar As String
    Dim CurStateCodeSource As String
    Dim CurStateDescriptionSource As String
    
    Dim CurStateCodeTarget As String
    Dim CurStateDescriptionTarget As String
    
    Dim CurActionToTake As String
    
    Dim CurCountUpdate As Long
    
    MaxRows = Cur.Range("tblState").Rows.Count
    For i = 1 To MaxRows
        CurActionToTake = Trim(Cur.Range("tblState[Action To Take]").Rows(i))
            'Update
            If CurActionToTake = "Update" Then
            
                If CurActionToTake = "Update" Then CurCountUpdate = CurCountUpdate + 1
                
                CurFacTar = Trim(Cur.Range("tblState[FacilityMnem_Target]").Rows(i))
                CurStateCodeSource = Trim(Cur.Range("tblState[State_Code_Source]").Rows(i))
                
                CurKey = CurFacTar & vbTab & CurStateCodeSource
                
                CurStateDescriptionSource = Trim(Cur.Range("tblState[State_Description_Source]").Rows(i))
                CurStateCodeTarget = Trim(Cur.Range("tblState[State_Code_Target]").Rows(i))
                CurStateDescriptionTarget = Trim(Cur.Range("tblState[State_Description_Target]").Rows(i))
                
                CurValue = CurStateDescriptionSource & vbTab & CurStateCodeTarget & vbTab & CurStateDescriptionTarget
                           
                If dUpdateItem.Exists(CurKey) = False Then
                    dUpdateItem.Add CurKey, CurValue
                End If
            End If
          
    Next i
    
    Dim UserResponse As Long
    If dUpdateItem.Count > 0 Then
    
        UserConfirmationText = "You are about to ..."
        If CurCountUpdate > 0 Then UserConfirmationText = UserConfirmationText & Chr(13) & "Update: " & CStr(CurCountUpdate)
        
        UserConfirmationText = UserConfirmationText & Chr(13) & Chr(13) & "Are you sure?"
    
        UserResponse = MsgBox(UserConfirmationText, vbYesNoCancel, "User Confirmation")
        If UserResponse <> vbYes Then
            MsgBox "User Cancelled. Nothing happened."
            Exit Sub
        End If
       
        Call UpdateMapping_State(dUpdateItem)
    End If

    Call RefreshData_State

End Sub

Private Sub UpdateMapping_State(dUpdateItem As Scripting.Dictionary)
    Dim CurFacTar As String
    Dim CurStateCodeSource As String
    Dim CurStateDescriptionSource As String
    
    Dim CurStateCodeTarget As String
    Dim CurStateDescriptionTarget As String
    
    Dim CurActionToTake As String

    Call OpenConnection

    Dim Rs As ADODB.Recordset
    Dim Cmd As New ADODB.Command
    For Each myKey In dUpdateItem
    
        KeyArray = Split(myKey, vbTab)
        CurFacTar = KeyArray(0)
        CurStateCodeSource = KeyArray(1)
       
        ValueArray = Split(dUpdateItem(myKey), vbTab)
        CurStateDescriptionSource = ValueArray(0)
        CurStateCodeTarget = ValueArray(1)
        CurStateDescriptionTarget = ValueArray(2)
        
        Set Rs = New ADODB.Recordset
        Set Cmd = Nothing
        Cmd.CommandText = "[dbo].[UI_Excel_Update_Xwalk_NonMagic_State]"
        Cmd.CommandType = adCmdStoredProc
        Cmd.ActiveConnection = Cn
        Set prm = Cmd.CreateParameter("@facmnem_tar", adVarChar, adParamInput, 25, CurFacTar)
        Cmd.Parameters.Append (prm)
        Set prm = Cmd.CreateParameter("@statecode_src", adVarChar, adParamInput, 25, CurStateCodeSource)
        Cmd.Parameters.Append (prm)
        Set prm = Cmd.CreateParameter("@statedesc_src", adVarChar, adParamInput, 255, CurStateDescriptionSource)
        Cmd.Parameters.Append (prm)
        Set prm = Cmd.CreateParameter("@statecode_tar", adVarChar, adParamInput, 25, CurStateCodeTarget)
        Cmd.Parameters.Append (prm)
        Set prm = Cmd.CreateParameter("@statedesc_tar", adVarChar, adParamInput, 255, CurStateDescriptionTarget)
        Cmd.Parameters.Append (prm)
        Set prm = Cmd.CreateParameter("@usersec34", adVarChar, adParamInput, 255, CurUser34)
        Cmd.Parameters.Append (prm)
       
        Set Rs = Cmd.Execute
    
        CurRecordCount = CurRecordCount + 1
        If LastOperationDateTime = Empty Then LastOperationDateTime = Now
        If Now - LastOperationDateTime > ((1 / 24 / 60 / 60) * 1) Then
            Call Utility_ProgressBar(CDec(CurRecordCount), dUpdateItem.Count)
            LastOperationDateTime = Now
        End If
    
    Next myKey
    
    Application.ScreenUpdating = True
    Application.StatusBar = False
    DoEvents
    
    If Rs.State <> 0 Then Rs.Close
    Call CloseConnection

End Sub

Sub Helper_FormatTopSectionOfCrosswalks(FirstColumnRange As Range, LegacyRange As Range, ExpanseRange As Range)

    With FirstColumnRange.Interior
        .Pattern = xlSolid
        .PatternColorIndex = xlAutomatic
        .ThemeColor = xlThemeColorDark2
        .TintAndShade = 0
        .PatternTintAndShade = 0
    End With

    With LegacyRange
            .Merge
            .Interior.Pattern = xlSolid
            .Interior.PatternColorIndex = xlAutomatic
            .Interior.Color = 5296274
            .Interior.TintAndShade = 0
            .Interior.PatternTintAndShade = 0
            .Font.Bold = True
            .HorizontalAlignment = xlCenter
    End With
    
    With ExpanseRange
            .Merge
            .Interior.Pattern = xlSolid
            .Interior.PatternColorIndex = xlAutomatic
            .Interior.ThemeColor = xlThemeColorAccent2
            .Interior.TintAndShade = 0.399975585192419
            .Interior.PatternTintAndShade = 0
            .Font.Bold = True
            .HorizontalAlignment = xlCenter
    End With

Sub SortWorksheetsTabs(CurWkBk As Workbook)
Application.ScreenUpdating = False
Dim ShCount As Integer, i As Integer, j As Integer
ShCount = CurWkBk.Sheets.Count

For i = 1 To ShCount - 1
    For j = i + 1 To ShCount
        If UCase(CurWkBk.Sheets(j).Name) < UCase(CurWkBk.Sheets(i).Name) Then
            CurWkBk.Sheets(j).Move before:=CurWkBk.Sheets(i)
        End If
    Next j
Next i

CurWkBk.Sheets("Table of Contents").Move before:=CurWkBk.Sheets(1)
Application.ScreenUpdating = True
End Sub
